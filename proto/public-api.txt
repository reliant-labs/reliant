# The PUBLIC Reliant API: which rpcs the Go SDK (sdk/go) and its generated
# clients expose.
#
# Modelled on control-plane's proto/public-api.txt, at RPC granularity because
# services here are mixed (TriggerService has FireTrigger and EnableWorkflow).
#
# EVERY service under proto/reliant/v1 appears exactly once, as
#
#   export   <Service>             all of its rpcs are public
#   partial  <Service>             only the `export <Service>/<Rpc>` lines below
#   withhold <Service>             none is public
#
# and every `export <Service>/<Rpc>` names a real rpc of a `partial` service.
# internal/buildmode/sdk_public_api_test.go fails on an unclassified service, so
# nothing becomes public by default. It also checks that buf.gen.yaml's SDK
# `types:` list, the SDK-only M mappings and the generated tree in sdk/go/gen
# all say exactly what this file says.
#
# NEVER publish (the test refuses them even if listed): the activation rpcs
# (EnableWorkflow, DisableWorkflow, SetTriggerEnabled, CreateTrigger,
# UpdateTrigger, DeleteTrigger, RotateWebhookToken), token minting
# (TokenService), the chat's daemon wiring (SetChatDaemon) and every
# daemon-facing service.
#
# Stability: PauseRun, ResumeRun and InterruptRun are beta (their semantics
# are still settling); the rest is stable within reliant.v1.

# --- Runs: start, observe and steer an execution. ---------------------------
export   RunService

# --- Triggers: read an automation, fire it, and follow what it launched. ----
partial  TriggerService
export   TriggerService/GetTrigger
export   TriggerService/ListTriggers
export   TriggerService/FireTrigger
export   TriggerService/ListTriggerEvents
export   TriggerService/GetLaunchEvent

# --- What following a run needs. -------------------------------------------
# A run's progress is the update feed of the session it lives in, and its open
# gates are that session's questions and approvals. The SDK reads them to follow
# a run, list its gates and answer a question. No Chat verb is exposed.
partial  ChatService
export   ChatService/GetChatUpdates
export   ChatService/GetWorkflowExecutions
# The bodies a TOOL_BODY_MODE_STUB read left out (GetChatUpdatesRequest's
# tool_bodies), by block id. Owner decision (2026-11-08): exported.
export   ChatService/GetContentBlockBodies

partial  QuestionService
export   QuestionService/ListPendingQuestions
export   QuestionService/ResolveQuestion

# Owner decision (2026-11-06): the SDK may decide approvals. The server requires
# the approvals:decide scope and refuses a token bound to a machine (daemon).
partial  ApprovalService
export   ApprovalService/ListApprovalsByChat
export   ApprovalService/Approve
export   ApprovalService/Deny
export   ApprovalService/BatchApprove
export   ApprovalService/BatchDeny

# --- Inbox: what is waiting on the caller. -----------------------------------
# CleanupStorage and ForgetStorageWorktree manage the machine and stay out.
# inbox.proto imports tools_daemon.proto, so that file's MESSAGE types are
# generated into the SDK too; none of ToolsDaemonService's rpcs is public.
partial  InboxService
export   InboxService/ListInbox
export   InboxService/DismissInboxItem
export   InboxService/RestoreInboxItem
